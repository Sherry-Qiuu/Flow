## 2. 上电顺序（每次都照这个来）

```
1. 检查外部：支架合上、无杂物压住散热孔、线没被夹住
2. 打开电池开关（如果有）
3. 树莓派红灯亮、绿灯闪
4. 屏幕亮起（约 20–40 秒）
5. 进入产品界面
6. 自检报告出现在屏幕上（10 秒内）
```

**关机顺序**（比开机更重要）：

```bash
sudo shutdown -h now
```

等绿灯**完全停止闪烁**（约 15 秒）再断电池开关。**直接拔电 = 一定会有一天把 SD 卡写坏。**

## 3. 整机自检脚本（核心产出）

把所有章节的验证代码合成一个文件。这个文件就是软件侧 `T02 硬件抽象层` 的起点。

`~/flow/code/self_test.py`：

```python
#!/usr/bin/env python3
"""心流 FLOW · 整机硬件自检
用法：python3 self_test.py [--json]
"""
import json
import os
import subprocess
import sys
from pathlib import Path

RESULTS = []


def record(name: str, ok: bool, detail: str = ""):
    RESULTS.append({"item": name, "ok": bool(ok), "detail": detail})
    mark = "✅" if ok else "❌"
    print(f"{mark}  {name:<18} {detail}")


# ---------- 1. 电源 ----------
def check_power():
    try:
        v = subprocess.run(["vcgencmd", "pmic_read_adc", "EXT5V_V"],
                           capture_output=True, text=True).stdout
        volt = None
        for tok in v.replace("=", " ").split():
            try:
                f = float(tok)
                if 3.0 < f < 6.0:
                    volt = f
            except ValueError:
                pass
        ok = volt is not None and volt >= 4.80
        record("5V 供电电压", ok, f"{volt:.2f} V" if volt else "读取失败")
    except FileNotFoundError:
        record("5V 供电电压", False, "vcgencmd 不存在")

    try:
        t = subprocess.run(["vcgencmd", "get_throttled"],
                           capture_output=True, text=True).stdout.strip()
        ok = t.endswith("0x0")
        record("欠压/降频", ok, t.split("=")[-1])
    except Exception:
        record("欠压/降频", False, "读取失败")

    try:
        temp = subprocess.run(["vcgencmd", "measure_temp"],
                              capture_output=True, text=True).stdout
        deg = float(temp.split("=")[1].split("'")[0])
        record("CPU 温度", deg < 80, f"{deg:.1f} °C")
    except Exception:
        record("CPU 温度", False, "读取失败")


# ---------- 2. I2C 设备 ----------
def check_i2c():
    try:
        from smbus2 import SMBus
        bus = SMBus(1)
        part = bus.read_byte_data(0x73, 0x00)
        record("手势传感器", part == 0x20, f"I2C 0x73, PartID=0x{part:02X}")
        bus.close()
    except Exception as e:
        record("手势传感器", False, str(e)[:40])


# ---------- 3. 摄像头 ----------
def check_camera():
    try:
        from camera import probe, capture
        if not probe():
            record("摄像头", False, "未检测到")
            return
        p = capture(timeout_ms=1000)
        size_kb = os.path.getsize(p) / 1024
        record("摄像头", size_kb > 50, f"{size_kb:.0f} KB -> {Path(p).name}")
    except Exception as e:
        record("摄像头", False, str(e)[:40])


# ---------- 4. 数字输入 ----------
def check_inputs():
    try:
        from inputs import ButtonInput, PhonePresence
        b = ButtonInput()
        record("STOP 按钮", True, f"当前{'按下' if b.pressed else '未按下'}")
        b.close()
    except Exception as e:
        record("STOP 按钮", False, str(e)[:40])

    try:
        from inputs import PhonePresence
        h = PhonePresence()
        record("手机在位(霍尔)", True, f"当前{'在位' if h.present else '不在位'}")
        h.close()
    except Exception as e:
        record("手机在位(霍尔)", False, str(e)[:40])


# ---------- 5. 音频 ----------
def check_audio():
    try:
        out = subprocess.run(["aplay", "-l"], capture_output=True, text=True).stdout
        has_speaker = "card" in out
        record("扬声器", has_speaker, "已识别" if has_speaker else "未识别")
    except FileNotFoundError:
        record("扬声器", False, "aplay 不存在")

    try:
        out = subprocess.run(["arecord", "-l"], capture_output=True, text=True).stdout
        has_mic = "card" in out
        record("麦克风", has_mic, "已识别" if has_mic else "未识别")
    except FileNotFoundError:
        record("麦克风", False, "arecord 不存在")


# ---------- 6. 网络 ----------
def check_network():
    try:
        r = subprocess.run(["ping", "-c", "1", "-W", "2", "223.5.5.5"],
                           capture_output=True, text=True)
        ok = r.returncode == 0
        record("网络连通", ok, "在线" if ok else "离线（进入降级模式）")
    except Exception:
        record("网络连通", False, "检测失败")


def main():
    print("=" * 46)
    print(" 心流 FLOW · 硬件自检")
    print("=" * 46)
    check_power()
    check_i2c()
    check_camera()
    check_inputs()
    check_audio()
    check_network()
    print("=" * 46)

    bad = [r for r in RESULTS if not r["ok"]]
    print(f"通过 {len(RESULTS)-len(bad)} / {len(RESULTS)}")

    if bad:
        print("\n需要处理：")
        for r in bad:
            print(f"  - {r['item']}: {r['detail']}")

    if "--json" in sys.argv:
        out = Path.home() / "flow" / "data" / "device_status.json"
        out.write_text(json.dumps(RESULTS, ensure_ascii=False, indent=2))
        print(f"\n已写入 {out}")

    return 0 if not bad else 1


if __name__ == "__main__":
    sys.exit(main())
```

运行：

```bash
cd ~/flow/code
python3 self_test.py --json
```

> **注意**：`网络连通` 失败不算故障 —— 离线降级是设计内的。真正要关心的是前五项。

## 4. 开机自启（15 分钟）

用 systemd 让自检和主程序开机就跑。

`/etc/systemd/system/flow-selftest.service`：

```ini
[Unit]
Description=FLOW hardware self-test
After=multi-user.target

[Service]
Type=oneshot
User=flow
WorkingDirectory=/home/flow/flow/code
ExecStart=/usr/bin/python3 /home/flow/flow/code/self_test.py --json
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target
```

启用：

```bash
sudo systemctl daemon-reload
sudo systemctl enable flow-selftest.service
sudo systemctl start flow-selftest.service
systemctl status flow-selftest.service
```

> 软件侧 `T01/T03` 会把 `flow-backend.service` 和 `flow-ui.service` 也加进来，形成三段启动：自检 → 后端 → 界面。

## 7. 整机排错顺序

出问题时**按这个顺序查**，不要乱拆：

```
1. 电源      → 电压对不对？有没有欠压？
2. 连接      → 排线/杜邦线有没有松？（80% 的问题在这里）
3. 单模块    → 单独测那个模块还行不行？
4. 干扰      → 是不是别的模块在抢电/抢总线？
5. 软件      → 驱动/权限/配置
6. 结构      → 装壳后才有 → 拆开测
```

详细的"现象 → 原因 → 处理"在 [B · 故障排查手册](B-故障排查手册.md)。
